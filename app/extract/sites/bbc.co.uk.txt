# Promotion banners (a wide strip of a picture with no caption whose description says it is a
# banner, or BBC Verify's logo), and the lines they head: a newsletter's sign-up (a link to /newsletters/),
# BBC Africa's "Go to BBCAfrica.com… Follow us", "What do you want BBC Verify to investigate?".
strip: //div[@data-block = 'image'][not(.//figcaption)][.//img[@alt = 'BBC Verify logo' or ((contains(@alt, 'banner') or contains(@alt, 'Banner')) and number(@width) > 3 * number(@height))]]
strip: //div[@data-block = 'text'][.//a[contains(@href, '/newsletters/')]][string-length(normalize-space()) < 400]
strip: //div[@data-block = 'text'][starts-with(normalize-space(), 'Go to BBCAfrica.com')]
strip: //div[@data-block = 'text'][normalize-space() = 'What do you want BBC Verify to investigate?']
strip: //p[contains(normalize-space(), 'you can get in touch via BBC Your Voice')][string-length(normalize-space()) < 200]
# The "Get in touch" form ("Are you personally affected…?"), the topic tags, story lists and
# event promotions; a breaking story's two closing lines about how to get breaking news.
strip: //div[@data-block = 'uploaderEmbed' or @data-block = 'topicList' or @data-block = 'promoList' or @data-block = 'links' or @data-block = 'eventPromo']
# Cards for other BBC stories (each with a "www.bbc.co.uk" line under it).
strip: //*[@data-testid = 'links-grid']
strip: //p[starts-with(normalize-space(), 'This breaking news story is being updated')]
strip: //p[starts-with(normalize-space(), 'You can receive Breaking News on a smartphone')]
# bbc.com's other layout (styled components, no data-block): the same banners (their pictures
# have no size, so by their description), newsletter lines, the "get in touch" line, topic tags.
strip: //figure[not(.//figcaption)][.//div[@data-testid = 'image']//img[contains(@alt, 'banner promoting') or starts-with(@alt, 'Your Voice banner') or starts-with(@alt, 'Thin, ')]]
strip: //p[contains(@class, 'Paragraph-styles__')][.//a[contains(@href, '/newsletters/')]][string-length(normalize-space()) < 400]
strip: //p[contains(@class, 'Paragraph-styles__')][starts-with(normalize-space(), 'If it is safe to do so, you can also get in touch')]
strip: //div[contains(@class, 'TagList-styles__')]
# Video pages and iPlayer and Sounds episodes.
not_article: ^/news/videos/
not_article: ^/(iplayer|sounds)/
