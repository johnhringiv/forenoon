# The article's box with its lead picture (only on a page that has the box: a feature's layout
# without one is left to readability). On a live blog the box holds each post with its heading,
# which readability drops with the post's header.
body: //div[@data-gu-name = 'body'] | //div[@data-gu-name = 'media'][//div[@data-gu-name = 'body']]
# Inside it: videos (a caption without its video: YouTube in the text, YouTube or self-hosted as
# the lead), cards for other Guardian stories, the newsletter sign-up and reporter-callout
# boxes and newsletter lines, the long read's podcast footer; the relative times ("2h ago") of a
# live blog's posts and of its "Key events" list (the list stays: on a long blog it is the only
# outline of the posts on later pages); a pinned post whose original is on this page.
strip: //figure[contains(@data-spacefinder-type, 'YoutubeBlockElement')]
strip: //div[@data-gu-name = 'media']//figure[.//gu-island[@name = 'YoutubeBlockComponent' or @name = 'SelfHostedVideo']]
strip: //*[contains(@data-spacefinder-type, 'RichLinkBlockElement')]
strip: //*[contains(@data-spacefinder-type, 'NewsletterSignupBlockElement')]
strip: //*[contains(@data-spacefinder-type, 'ReporterCalloutBlockElement')]
strip: //*[self::p or self::li][.//a[contains(@href, 'CMP=copyembed')]]
strip: //p[starts-with(normalize-space(), 'Listen to our podcasts here and sign up to the long read weekly email')]
strip: //gu-island[@name = 'KeyEventsCarousel']//time
strip: //div[@id = 'pinned-post'][(.//a[contains(@href, '#block-')])[1][substring-after(@href, '#') = //div[@id = 'liveblog-body']/article/@id]]
strip: //gu-island[@name = 'RelativeTime']
# A live post's contributor portrait, in the post's header.
strip: //article[starts-with(@id, 'block-')]/header//img
# A lead picture's caption shown on a tap over the photo, a copy of the caption under it.
strip: //div[@id = 'the-caption']
# The topic tags after the article ("Explore more on these topics").
strip: //div[span[normalize-space() = 'Explore more on these topics']]
# Podcast episodes and video pages.
not_article: /(audio|video)/\d{4}/
